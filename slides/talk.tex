\begin{frame}[fragile]{Dangerous mounts}
\begin{lstlisting}
Path("/shared/next-job.json").write_text('{"queue":"attacker"}')

writable bind   WRITE_SUCCEEDED   host file changed
read-only bind  WRITE_DENIED      host file unchanged

Path("/output/patch.txt").write_text("legitimate patch output")
# succeeds in both cases
\end{lstlisting}
\subline{Capability-scoped filesystems: /input ro, /output rw.}
\evidence
\note{03:20 to 05:00. We mounted another job directory writable. The write is a normal open. The host file changes. Read-only bind denies the same write. Output still works. If you snapshot this disk and fork it, the next agent inherits the same files.}
\end{frame}

\begin{frame}[fragile]{Exfiltration through an allowed endpoint}
\begin{lstlisting}
POST /accounts/other/uploads

# extra keys -> DENIED, no delivery
# broker POST /accounts/team/reports
#        {"task":"parser-fix","status":"completed"}
\end{lstlisting}
\subline{Process isolation: worker cannot reach the service. Broker owns the POST.}
\evidence
\note{05:00 to 07:00. We allowed the report host. The worker posted the private fixture to another account on that host. 201. The worker does not choose URL, recipient, or body. Extra keys denied. Direct route gone. Team report still arrives.}
\end{frame}

\begin{frame}[fragile]{Verifier tampering}
\begin{lstlisting}
# checker.py  hash unchanged
expected = json.loads(Path("expected.json").read_text())

# worker-written expected.json
{"input": "admin:none", "output": 200}

python3 handler.py admin:none
200     # shipped
401     # required
CI      PASS
\end{lstlisting}
\subline{The worker edited the cases. The checker file did not change.}
\evidence
\note{07:00 to 09:45. Ticket: unauthenticated /admin returns 401. checker.py is read-only. It loads expected.json from the workspace. Worker writes admin:none -> 200. Checker prints PASS. Independent case still requires 401. Shipped handler prints 200. A second machine that reads the same expected.json also prints PASS. Ignore a worker PASS file; that is the detector.}
\end{frame}

\begin{frame}[fragile]{Close each one}
\begin{lstlisting}

\end{lstlisting}
\subline{History: rewritten log => different chain head. The compare decides merge.}
\evidence
\note{09:45 to 11:45. Read the four abstract locks as you walk the snippet. Task token 403 on the other project. Read-only input. Worker cannot open the service. Freeze handler.py, compare on the controller, publish that digest. Used nonce cannot replay. Open admin is not published. Locked admin is published. The log does not decide the merge.}
\end{frame}
